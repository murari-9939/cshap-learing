    public static int[] twoSum(int[] nums, int target) {
        // Using a HashMap to store value and index
        Map<Integer, Integer> map = new HashMap<>();
        
        // Iterating through the array
        for (int i = 0; i < nums.length; i++) {
            int complement = target - nums[i];  // Find the complement
            
            // If the complement exists in the map, return the indices
            if (map.containsKey(complement)) {
                return new int[] { map.get(complement), i };
            }
            
            // If not, store the current number and its index in the map
            map.put(nums[i], i);
        }

        // Return an empty array if no solution found (though it's assumed that there's always one)
        return new int[] {};
    }
}

// Define a Node class
class Node {
    int data;        // Data part of the node
    Node next;       // Pointer to the next node

    public Node(int data) {
        this.data = data;
        this.next = null;
    }
}

// Define the LinkedList class
class LinkedList {
    Node head;       // Head of the list

    // Constructor
    public LinkedList() {
        head = null;
    }

    // Method to add a new node at the end of the list
    public void add(int data) {
        Node newNode = new Node(data);  // Create a new node
        if (head == null) {             // If the list is empty
            head = newNode;             // The new node becomes the head
        } else {
            Node current = head;        // Start from the head
            while (current.next != null) { // Traverse till the end of the list
                current = current.next;
            }
            current.next = newNode;     // Append the new node
        }
    }

    // Method to delete a node by its value
    public void delete(int data) {
        if (head == null) {
            System.out.println("List is empty.");
            return;
        }

        if (head.data == data) {        // If the node to be deleted is the head
            head = head.next;           // Change head to the next node
            return;
        }

        Node current = head;
        Node previous = null;
        while (current != null && current.data != data) { // Search for the node
            previous = current;
            current = current.next;
        }

        if (current == null) {          // If the node was not found
            System.out.println("Node not found.");
            return;
        }

        previous.next = current.next;   // Unlink the node from the list
    }

    // Method to display the list
    public void display() {
        if (head == null) {
            System.out.println("List is empty.");
            return;
        }

        Node current = head;
        while (current != null) {
            System.out.print(current.data + " -> ");
            current = current.next;
        }
        System.out.println("null");
    }
}

public class Stack {
    private int[] stackArray; // Array to store stack elements
    private int top; // Index of the top element
    private int capacity; // Maximum size of the stack

    // Constructor to initialize the stack with a specific capacity
    public Stack(int size) {
        stackArray = new int[size]; // Create an array with the given size
        capacity = size; // Set the maximum size
        top = -1; // Initialize the stack as empty
    }

    // Push method to add an element to the top of the stack
    public void push(int value) {
        if (isFull()) {
            System.out.println("Stack Overflow! Unable to add " + value);
            return;
        }
        stackArray[++top] = value; // Increment top and add value
    }

    // Pop method to remove and return the top element from the stack
    public int pop() {
        if (isEmpty()) {
            System.out.println("Stack Underflow! No elements to pop.");
            return -1; // Return an error value (could also throw an exception)
        }
        return stackArray[top--]; // Return the top element and decrement top
    }

    // Peek method to return the top element without removing it
    public int peek() {
        if (isEmpty()) {
            System.out.println("Stack is empty.");
            return -1; // Return an error value (could also throw an exception)
        }
        return stackArray[top]; // Return the top element
    }

    // Method to check if the stack is empty
    public boolean isEmpty() {
        return top == -1; // True if top is -1, indicating no elements
    }

    // Method to check if the stack is full
    public boolean isFull() {
        return top == capacity - 1; // True if top is at the last index of the array
    }

    // Method to get the current size of the stack
    public int size() {
        return top + 1; // The size is top index + 1
    }

    public static void main(String[] args) {
        // Create a stack with a capacity of 5
        Stack stack = new Stack(5);

        // Push elements to the stack
        stack.push(10);
        stack.push(20);
        stack.push(30);
        stack.push(40);
        stack.push(50);

        // Trying to push when the stack is full
        stack.push(60); // Will show "Stack Overflow!"

        // Pop elements from the stack
        System.out.println("Popped: " + stack.pop()); // 50
        System.out.println("Popped: " + stack.pop()); // 40

        // Peek at the top element
        System.out.println("Top element: " + stack.peek()); // 30

        // Check if the stack is empty
        System.out.println("Is stack empty? " + stack.isEmpty()); // False

        // Push a new element
        stack.push(60);

        // Show the current size of the stack
        System.out.println("Stack size: " + stack.size()); // 4
    }
}

public class Queue {
    private int[] queueArray; // Array to store queue elements
    private int front; // Index of the front element
    private int rear; // Index of the rear element
    private int capacity; // Maximum size of the queue
    private int size; // Current size of the queue

    // Constructor to initialize the queue with a specific capacity
    public Queue(int size) {
        queueArray = new int[size]; // Create an array with the given size
        capacity = size; // Set the maximum size
        front = 0; // Front starts at index 0
        rear = -1; // Rear starts at -1 (before the first element)
        this.size = 0; // Initially, the queue is empty
    }

    // Enqueue method to add an element to the rear of the queue
    public void enqueue(int value) {
        if (isFull()) {
            System.out.println("Queue Overflow! Unable to add " + value);
            return;
        }
        rear = (rear + 1) % capacity; // Circular increment of rear
        queueArray[rear] = value; // Add value to the rear of the queue
        size++; // Increase the size
    }

    // Dequeue method to remove and return the front element from the queue
    public int dequeue() {
        if (isEmpty()) {
            System.out.println("Queue Underflow! No elements to dequeue.");
            return -1; // Return an error value (could also throw an exception)
        }
        int dequeuedValue = queueArray[front]; // Get the front element
        front = (front + 1) % capacity; // Circular increment of front
        size--; // Decrease the size
        return dequeuedValue;
    }

    // Peek method to return the front element without removing it
    public int peek() {
        if (isEmpty()) {
            System.out.println("Queue is empty.");
            return -1; // Return an error value (could also throw an exception)
        }
        return queueArray[front]; // Return the front element
    }

    // Method to check if the queue is empty
    public boolean isEmpty() {
        return size == 0; // True if size is 0, indicating no elements
    }

    // Method to check if the queue is full
    public boolean isFull() {
        return size == capacity; // True if the queue size equals its capacity
    }

    // Method to get the current size of the queue
    public int getSize() {
        return size; // Return the current number of elements
    }

    public static void main(String[] args) {
        // Create a queue with a capacity of 5
        Queue queue = new Queue(5);

        // Enqueue elements into the queue
        queue.enqueue(10);
        queue.enqueue(20);
        queue.enqueue(30);
        queue.enqueue(40);
        queue.enqueue(50);

        // Trying to enqueue when the queue is full
        queue.enqueue(60); // Will show "Queue Overflow!"

        // Dequeue elements from the queue
        System.out.println("Dequeued: " + queue.dequeue()); // 10
        System.out.println("Dequeued: " + queue.dequeue()); // 20

        // Peek at the front element
        System.out.println("Front element: " + queue.peek()); // 30

        // Check if the queue is empty
        System.out.println("Is queue empty? " + queue.isEmpty()); // False

        // Enqueue a new element after dequeuing
        queue.enqueue(60);

        // Show the current size of the queue
        System.out.println("Queue size: " + queue.getSize()); // 4
    }
}

    {
        int n = nums.Count; // Use Count for List instead of Length for array

        // Check for the first element
        if (n == 1 || nums[0] > nums[1])
            return 0;

        // Check for the last element
        if (nums[n - 1] > nums[n - 2])
            return n - 1;

        // Check for peaks in the middle
        for (int i = 1; i < n - 1; i++)
        {
            if (nums[i] > nums[i - 1] && nums[i] > nums[i + 1])
                return i;
        }

        return -1; // This line should never be reached if there's always a peak
    }

        String ste = "murarmmmkkkjjjjj"; // Input string
        
        HashMap<Character, Integer> dt = new HashMap<Character, Integer>(); 
        
        // Loop through each character in the string
        for (char c : ste.toCharArray()) {
            // If the character is already in the map, increment its count
            if (dt.containsKey(c)) {
                dt.put(c, dt.get(c) + 1);
            } else {
                // Otherwise, add it to the map with a count of 1
                dt.put(c, 1);
            }
        }

        // Print the character counts
        for (char key : dt.keySet()) {
            System.out.println(key + ": " + dt.get(key));

import java.util.HashMap;

public class WordCount {
    public static void main(String[] args) {
        String str = "the day is sunny the the\r\n" +
                     "the sunny is is";
        
        HashMap<String, Integer> wordCountMap = new HashMap<String, Integer>(); 
        
        // Split the string into words using whitespace as a delimiter
        String[] words = str.split("\\s+");
        
        // Loop through each word
        for (String word : words) {
            // Remove punctuation and convert to lower case for consistency
            word = word.trim().toLowerCase();
            
            // If the word is already in the map, increment its count
            if (wordCountMap.containsKey(word)) {
                wordCountMap.put(word, wordCountMap.get(word) + 1);
            } else {
                // Otherwise, add it to the map with a count of 1
                wordCountMap.put(word, 1);gf
            }
        }

        // Print the word counts
        for (String key : wordCountMap.keySet()) {
            System.out.println(key + ": " + wordCountMap.get(key));
        }
    }
}
