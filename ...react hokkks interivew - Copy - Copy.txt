"use client";
import { useState } from "react";

export default function Calculator() {
  const [num1, setNum1] = useState(0);
  const [num2, setNum2] = useState(0);
  const [operation, setOperation] = useState("+");
  const [result, setResult] = useState(0);

  const calculateResult = () => {
    switch (operation) {
      case "+": setResult(num1 + num2); break;
      case "-": setResult(num1 - num2); break;
      case "*": setResult(num1 * num2); break;
      case "%": setResult(num1 % num2); break;
      default: setResult(0);
    }
  };

  return (
    <div>
      <h2>Simple Calculator</h2>
      <input type="number" value={num1} onChange={e => setNum1(Number(e.target.value))} />
      <select onChange={e => setOperation(e.target.value)} value={operation}>
        <option value="+">+</option>
        <option value="-">-</option>
        <option value="*">*</option>
        <option value="%">%</option>
      </select>
      <input type="number" value={num2} onChange={e => setNum2(Number(e.target.value))} />
      <button onClick={calculateResult}>=</button>
      <h3>Result: {result}</h3>
    </div>
  );
}
🔹 Explanation
useState for num1, num2, operation, and result → Tracks values dynamically.
calculateResult function → Updates the result based on the selected operation.
Dropdown (<select>) → Lets users choose between +, -, *, and %.
Button (=) → Triggers calculation.

import { useState, useMemo } from "react";

export default function Calculator() {
  const [num1, setNum1] = useState(0);
  const [num2, setNum2] = useState(0);
  const [operation, setOperation] = useState("+");

  const result = useMemo(() => {
    switch (operation) {
      case "+": return num1 + num2;
      case "-": return num1 - num2;
      case "*": return num1 * num2;
      case "%": return num1 % num2;
      default: return 0;
    }
  }, [num1, num2, operation]);

  return (
    <div>
      <h2>Calculator</h2>
      <input type="number" value={num1} onChange={e => setNum1(Number(e.target.value))} />
      <select onChange={e => setOperation(e.target.value)} value={operation}>
        <option value="+">+</option>
        <option value="-">-</option>
        <option value="*">*</option>
        <option value="%">%</option>
      </select>
      <input type="number" value={num2} onChange={e => setNum2(Number(e.target.value))} />
      <h3>Result: {result}</h3>
    </div>
  );
}
🔹 Why Use useMemo?
It prevents unnecessary recalculations by only recomputing when num1, num2, or operation change.
More efficient for large calculations.

'use client'
import { useState } from "react";

// Define initial items separately
const initialItems = [
  { id: 1, name: "Laptop", price: 1000, quantity: 2 },
  { id: 2, name: "Mouse", price: 50, quantity: 5 },
  { id: 3, name: "Keyboard", price: 80, quantity: 3 }
];

export default function App() {
  // State to manage items
  const [items, setItems] = useState(initialItems);

  // Function to update quantity
  const updateQuantity = (id, newQuantity) => {
    setItems((prevItems) =>
      prevItems.map((item) =>
        item.id === id ? { ...item, quantity: newQuantity } : item
      )
    );
  };

  // Function to delete an item
  const deleteItem = (id) => {
    setItems((prevItems) => prevItems.filter((item) => item.id !== id));
  };

  return (
    <div>
      <ul>
        {items.map((e) => (
          <li key={e.id}>
            {e.name} - ${e.price} (Qty: {e.quantity}){" "}
            <button onClick={() => updateQuantity(e.id, e.quantity + 1)}>➕</button>
            <button onClick={() => updateQuantity(e.id, e.quantity - 1)}>➖</button>
            <button onClick={() => deleteItem(e.id)}>❌</button>
          </li>
        ))}
      </ul>
    </div>
  );
}
