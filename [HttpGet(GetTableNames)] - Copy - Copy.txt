[HttpGet("GetTableNames")]
        public IActionResult GetTableNames()
        {
            var connStr = this.Configuration.GetConnectionString("cpu52");

            using (SqlConnection con = new SqlConnection(connStr))
            {
                try
                {
                    con.Open();

                    SqlCommand sqlcmd = new SqlCommand("SELECT TableName, DisplayName FROM LPUOnlineExportMaster", con);
                    SqlDataAdapter sda = new SqlDataAdapter();
                    sda.SelectCommand = sqlcmd;
                    DataSet ds = new DataSet(); // Create a DataSet to store the result
                    sda.Fill(ds);
                    con.Close();

                    List<TableInfo> tableList = new List<TableInfo>();

                    if (ds.Tables.Count > 0)
                    {
                        if (ds.Tables[0].Rows.Count > 0)
                        {
                            foreach (DataRow dr in ds.Tables[0].Rows)
                            {
                                TableInfo Pk = new TableInfo();
                                Pk.TableName = dr["TableName"].ToString();
                                Pk.DisplayName = dr["DisplayName"].ToString();
                                tableList.Add(Pk);
                            }
                        }
                    }
                    else
                    {
                        // Handle the case where no data was returned
                        return NotFound("No data found");
                    }
                    return Ok(tableList);
                }
                catch (SqlException dp)
                {
                    return StatusCode(500, "SQL error has occurred: " + dp.Message);
                }
                catch (Exception ex)
                {
                    return StatusCode(500, "Error: " + ex.Message);
                }
            }
        }
