 CREATE TABLE Category (
    CategoryId INT IDENTITY(1,1) PRIMARY KEY, -- Auto-incremented unique ID for each category
    CategoryName NVARCHAR(100) NOT NULL,      -- Name of the category
    Description NVARCHAR(MAX) NULL,          -- Optional description of the category
    CreatedAt DATETIME DEFAULT GETDATE(),    -- Timestamp when the category is created
    UpdatedAt DATETIME DEFAULT GETDATE()     -- Timestamp when the category is updated
);
CREATE TABLE Subcategory (
    SubcategoryId INT IDENTITY(1,1) PRIMARY KEY, -- Auto-incremented unique ID for each subcategory
    SubcategoryName NVARCHAR(100) NOT NULL,      -- Name of the subcategory
    CategoryId INT NOT NULL,                     -- Foreign key referencing the Category table
    Description NVARCHAR(MAX) NULL,             -- Optional description of the subcategory
    CreatedAt DATETIME DEFAULT GETDATE(),       -- Timestamp when the subcategory is created
    UpdatedAt DATETIME DEFAULT GETDATE(),       -- Timestamp when the subcategory is updated
    CONSTRAINT FK_Subcategory_Category FOREIGN KEY (CategoryId)
    REFERENCES Category(CategoryId) ON DELETE CASCADE -- Cascade deletion if a category is deleted
);

CREATE PROCEDURE GetSubcategoriesByCategoryName
    @CategoryName NVARCHAR(100)
AS
BEGIN
    SELECT C.CategoryName, S.SubcategoryName, S.Description
    FROM Subcategory S
    INNER JOIN Category C ON S.CategoryId = C.CategoryId
    WHERE C.CategoryName = @CategoryName;
END;
