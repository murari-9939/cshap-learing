using System;

class Program
{
    static void Main()
    {
        int[][] arr1 = {
            new int[] {1, 6, 3, 4, 5},
            new int[] {1, 5, 3, 4, 56}
        };

        // Sorting each row using Bubble Sort
        for (int i = 0; i < arr1.Length; i++)
        {
            BubbleSort(arr1[i]);
        }

        // Printing the sorted array
        for (int i = 0; i < arr1.Length; i++)
        {
            for (int j = 0; j < arr1[i].Length; j++) 
            {
                Console.Write(arr1[i][j] + "\t");
            }
            Console.WriteLine();
        }
    }

    // Bubble Sort function to sort a single row (array)
    static void BubbleSort(int[] array)
    {
        int n = array.Length;
        for (int i = 0; i < n - 1; i++)
        {
            for (int j = 0; j < n - 1 - i; j++)
            {
                if (array[j] > array[j + 1])
                {
                    // Swap the elements
                    int temp = array[j];
                    array[j] = array[j + 1];
                    array[j + 1] = temp;
                }
            }
        }
    }
}

using System;

class Program
{
    static void Main()
    {
        int[,] arr = {
            {1, 2, 3},
            {4, 5, 6},
            {7, 8, 9}
        };

        int sum = 0;

        for (int i = 0; i < arr.GetLength(0); i++)         // rows
        {
            for (int j = 0; j < arr.GetLength(1); j++)     // columns
            {
                sum += arr[i, j];
            }
        }

        Console.WriteLine("Total sum = " + sum);
    }

class Node {
    int data;
    Node next;

    Node(int data) {
        this.data = data;
        this.next = null;
    }
}

public class Output {
    Node head; // Head of the linked list

    // Method to add a node to the linked list
    public void add(int data) {
        Node newNode = new Node(data);

        if (head == null) {
            head = newNode; // If list is empty, set head to the new node
        } else {
            Node current = head;
            while (current.next != null) {
                current = current.next; // Traverse to the end of the list
            }
            current.next = newNode; // Link the last node to the new node
        }
    }

    // Method to display the linked list
    public void display() {
        Node current = head;
        while (current != null) {
            System.out.print(current.data + " -> ");
            current = current.next;
        }
        System.out.println("null"); // End of the list
    }

    public static void main(String[] args) {
        Output list = new Output();

        // Add some nodes
        list.add(10);
        list.add(20);
        list.add(30);
        list.add(40);

        // Display the linked list
        System.out.println("Linked List:");
        list.display();
    }
}

public class Output {
    Node head; // Head of the linked list

    // Method to add a node to the end of the linked list
    public void add(int data) {
        Node newNode = new Node(data);

        if (head == null) {
            head = newNode; // If list is empty, set head to the new node
        } else {
            Node current = head;
            while (current.next != null) {
                current = current.next; // Traverse to the end of the list
            }
            current.next = newNode; // Link the last node to the new node
        }
    }

    // Method to display the linked list
    public void display() {
        Node current = head;
        while (current != null) {
            System.out.print(current.data + " -> ");
            current = current.next;
        }
        System.out.println("null"); // End of the list
    }

    // Method to get the size of the linked list
    public int size() {
        int count = 0;
        Node current = head;
        while (current != null) {
            count++;
            current = current.next;
        }
        return count;
    }

    // Method to search for a value in the linked list
    public boolean search(int value) {
        Node current = head;
        while (current != null) {
            if (current.data == value) {
                return true; // Value found
            }
            current = current.next;
        }
        return false; // Value not found
    }

    // Method to insert a node at a specific position
    public void insertAt(int data, int position) {
        Node newNode = new Node(data);

        if (position == 0) { // Insert at the head
            newNode.next = head;
            head = newNode;
        } else {
            Node current = head;
            for (int i = 0; i < position - 1 && current != null; i++) {
                current = current.next;
            }

            if (current == null) {
                System.out.println("Position out of bounds");
            } else {
                newNode.next = current.next;
                current.next = newNode;
            }
        }
    }

    // Method to delete a node by value
    public void deleteByValue(int value) {
        if (head == null) {
            System.out.println("List is empty.");
            return;
        }

        if (head.data == value) { // If the head node has the value
            head = head.next;
            return;
        }

        Node current = head;
        while (current.next != null && current.next.data != value) {
            current = current.next;
        }

        if (current.next == null) {
            System.out.println("Value not found in the list.");
        } else {
            current.next = current.next.next; // Remove the node
        }
    }

    // Method to reverse the linked list
    public void reverse() {
        Node previous = null;
        Node current = head;
        Node next = null;
        while (current != null) {
            next = current.next; // Store the next node
            current.next = previous; // Reverse the current node's pointer
            previous = current; // Move pointers one position ahead
            current = next;
        }
        head = previous; // Set head to the last node (new head)
    }

import java.util.HashMap;

public class WordCount {
    public static void main(String[] args) {
        String str = "the day is sunny the the\r\n" +
                     "the sunny is is";
        
        HashMap<String, Integer> wordCountMap = new HashMap<String, Integer>(); 
        
        // Split the string into words using whitespace as a delimiter
        String[] words = str.split("\\s+");
        
        // Loop through each word
        for (String word : words) {
            // Remove punctuation and convert to lower case for consistency
            word = word.trim().toLowerCase();
            
            // If the word is already in the map, increment its count
            if (wordCountMap.containsKey(word)) {
                wordCountMap.put(word, wordCountMap.get(word) + 1);
            } else {
                // Otherwise, add it to the map with a count of 1
                wordCountMap.put(word, 1);
            }
        }

        // Print the word counts
        for (String key : wordCountMap.keySet()) {
            System.out.println(key + ": " + wordCountMap.get(key));
        }
    }
}

Word Counting:
