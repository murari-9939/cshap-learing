 CREATE PROCEDURE InsertStudent
    @FirstName VARCHAR(50),
    @LastName VARCHAR(50),
    @DateOfBirth DATE,
    @Gender VARCHAR(10),
    @Address VARCHAR(100),
    @ContactNumber VARCHAR(20),
    @Email VARCHAR(50)
AS
BEGIN
    INSERT INTO StudentMaster (
        FirstName, 
        LastName, 
        DateOfBirth, 
        Gender, 
        Address, 
        ContactNumber, 
        Email
    )
    VALUES (
        @FirstName, 
        @LastName, 
        @DateOfBirth, 
        @Gender, 
        @Address, 
        @ContactNumber, 
        @Email
    );
END;
